  % mean shape (solid)
  \draw[very thick] \liveroutline;
  % mode deformation arrows (double-headed = +-sigma), one per mode, on the real boundary,
  % colour-matched to the corresponding coeff cell b_i
  \draw[{Latex[length=1.1mm]}-{Latex[length=1.1mm]}, red!75,    thick] (0.30,-0.06) -- (0.55,-0.06); % mode 1 (b1), right flank

  \foreach \i/\c in {0/red!70, 1/orange!70, 2/teal!60}{
